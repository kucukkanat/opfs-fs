title: xterm.js
description: Build a durable browser terminal with xterm.js, just-bash, and opfs-fs.
---

`xterm.js` renders the terminal and supplies input. `just-bash` executes commands,
and `opfs-fs` owns the durable workspace. No xterm-specific filesystem adapter is
needed: adapt the workspace to `just-bash`, then connect xterm's `onData()` and
`write()` APIs to your shell session.

```sh
npm install @kucukkanat/opfs-fs@github:kucukkanat/opfs-fs#main just-bash @xterm/xterm @xterm/addon-fit react react-dom
```

```tsx
import { useEffect, useRef } from "react"
import { FitAddon } from "@xterm/addon-fit"
import { Terminal } from "@xterm/xterm"
import "@xterm/xterm/css/xterm.css"
import { Bash } from "just-bash/browser"
import { openWorkspace, type OpfsWorkspace } from "@kucukkanat/opfs-fs"
import {
  createJustBashFileSystem,
  createJustBashTerminalSession,
  type JustBashTerminalSession,
} from "@kucukkanat/opfs-fs/just-bash"

export function OpfsXterm() {
  const host = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const target = host.current
    if (!target) return
    const terminal = new Terminal({ cursorBlink: true, rows: 18 })
    const fit = new FitAddon()
    terminal.loadAddon(fit)
    terminal.open(target)
    fit.fit()

    let disposed = false
    let workspace: OpfsWorkspace | undefined
    let session: JustBashTerminalSession | undefined
    const write = (value: string): void => terminal.write(value)

    // This is the glue: xterm sends raw key data to the renderer-neutral
    // session, which edits the line and invokes bash.exec() when Enter arrives.
    const disposeInput = terminal.onData((data) => session?.handleInput(data, write))

    void (async () => {
      try {
        workspace = await openWorkspace({ name: "terminal", root: "/workspace" })
        if (disposed) { workspace.close(); return }
        const bash = new Bash({
          fs: createJustBashFileSystem(workspace),
          cwd: "/workspace",
          executionLimitProfile: "hardened",
        })
        session = createJustBashTerminalSession(bash, { cwd: "/workspace" })
        await workspace.writeFile("/workspace/welcome.txt", "Hello from OPFS\\n")
        write("opfs-fs + just-bash + xterm.js\\r\\n")
        session.writePrompt(write)
      } catch (error) {
        write(`Unable to open OPFS: ${error instanceof Error ? error.message : "Unknown error"}\\r\\n`)
      }
    })()

    return () => {
      disposed = true
      disposeInput.dispose()
      workspace?.close()
      terminal.dispose()
    }
  }, [])

  return <div ref={host} />
}
```

## Live example

<XtermDemo />

`createJustBashTerminalSession()` is the glue layer. It keeps command history, serializes commands, converts `\\n` to terminal line endings, and carries a standalone `cd` into the next command while leaving rendering to xterm.js. Type `help`, `pwd`, `ls`, `cat welcome.txt`, or `printf hello > note.txt`. Reload, then `cat note.txt` to verify OPFS persistence.

:::tip
`Bash.exec()` scopes `cwd` to one execution. `createJustBashTerminalSession()` retains the interactive current directory and passes it to each call.
:::
